### 5.10 엔티티 생명주기 (Simulation, IMP-LOOP) — 생성·회수의 유일한 경로

```csharp
namespace Survivor.Simulation
{
    /// Base component for pooled simulation entities (no FixedUpdate here).
    public abstract class SimEntity : MonoBehaviour
    {
        public EntityHandle Id { get; }
        public EntityCategory Category { get; }
        public CollisionBody Body { get; }
        public Vector2 Position { get; protected set; }  // authoritative; mirrors into Body.Position
        public long SpawnStep { get; }
        public long ExpireStep { get; }                  // long.MaxValue when no lifetime; LifetimeSystem despawns when Step >= ExpireStep
        public bool IsConsumed { get; }                  // (rev 2) set in Resolve; later events for this entity are ignored
        internal void Bind(EntityHandle id, long spawnStep, long expireStep);
        //   (rev 15) creates Body on the first Bind only (reused across pool cycles); every Bind sets Body.Id/Host and
        //   Body.Layer = the CollisionLayer with the same name as Category (R30)
        internal void MarkConsumed();
        internal void PlaceAt(Vector2 position);          // ★ (rev 15) Spawn: Position (and Body.Position) before initialize
        internal void SetCategory(EntityCategory category); // ★ (rev 15) PrefabCatalog only; Category is [SerializeField] so pool clones inherit it
        internal void Release();                          // ★ (rev 15) Despawn: OnReleased(), then Id = Body.Id = EntityHandle.None
        internal int ActiveSlot { get; set; }             // ★ (rev 15) slot in EntityLifecycle's active list, -1 while pooled
        protected abstract void OnReleased();             // clear references; called before pool release
    }

    /// Single owner of acquire/id/init/register and the exact reverse on despawn. Lifecycle phase or run reset only.
    public sealed class EntityLifecycle
    {
        public EntityLifecycle(IPoolStrategy pool, EntityHandleAllocator ids, ITickStrategy ticks,
                               ICollisionStrategy collisions, IPhysicsBodyRegistry physicsBodies,
                               PrefabCatalog prefabs, SimulationClock clock);
        public T Spawn<T>(EntityCategory category, Vector2 position, long lifetimeSteps, Action<T> initialize) where T : SimEntity;
        //   throws InvalidOperationException unless clock.CurrentPhase == Lifecycle or a run reset is in progress (TAC-0-8)
        //   (rev 5) lifetimeSteps == 0 -> no lifetime, ExpireStep = long.MaxValue (W2/E2 orbs); > 0 -> ExpireStep = SpawnStep + lifetimeSteps; < 0 -> ArgumentOutOfRangeException
        //   order: pool.Acquire -> ids.Allocate -> Bind -> initialize(entity) -> ticks.Register (if IUpdatable)
        //          -> physicsBodies.Register(Body) -> collisions.Register(Body) -> registry add (MonsterRegistry / PickupRegistry ...)
        //   (rev 15, code) pool.Acquire -> ids.Allocate -> Bind -> PlaceAt -> initialize -> ticks.Register (if IUpdatable)
        //          -> physicsBodies.Register(Body) -> collisions.Register(Body) -> active list add -> Spawned event (= "registry add")
        //   (rev 15) also throws InvalidOperationException when the category template's SimEntity is not a T
        public T Spawn<T>(EntityCategory category, GameObject prefabOverride, Vector2 position, long lifetimeSteps,
                          Action<T> initialize) where T : SimEntity;
        //   ★ (rev 15) same, from PrefabCatalog.GetTemplate(category, prefabOverride) (null = category template);
        //   the only path to an F11 prefab-override template. The overload above calls this with null.
        public bool Despawn(EntityHandle id, DespawnReason reason);
        //   same phase restriction; returns false if !ids.IsLive(id) (already gone; no exception)
        //   order: registry remove -> collisions.Unregister -> physicsBodies.Unregister -> ticks.Unregister
        //          -> Despawned event -> OnReleased -> ids.Free -> pool.Release
        //   (rev 15, code) Despawning event (= "registry remove") -> collisions.Unregister -> physicsBodies.Unregister
        //          -> ticks.Unregister (if IUpdatable) -> Despawned event -> Release (OnReleased, Id = None) -> active list remove
        //          -> ids.Free -> pool.Release
        public bool TryGet(EntityHandle id, out SimEntity entity);
        public void DespawnAll(DespawnReason reason);   // run reset, ascending Id order
        //   (rev 15) throws InvalidOperationException while a step is active; the private run-reset flag that lets Spawn/Despawn
        //   run outside Lifecycle is on only while DespawnAll runs [default]
        //   (rev 16) kept: the player is not a pooled entity and never goes through Spawn/Despawn (5.12 PlayerEntity, 5c.3 #1)
        public void ReplaceStrategies(IPoolStrategy pool, ITickStrategy ticks, ICollisionStrategy collisions); // run boundary only, after DespawnAll
        //   (rev 15) throws InvalidOperationException while a step is active or entities are live, ArgumentNullException for null.
        //   When pool is not the same object as the current pool: current (outgoing) pool.Clear() first, then swap (DECISIONS 2 2026-11-06).
        //   Same pool object kept -> no Clear(). Prewarming the incoming pool is App's job (PoolPrewarmCounts.PrewarmAll, 5.6).
        //   Logs "[Loop] Strategies replaced: ..." only when a Kind changed.
        public event Action<SimEntity, DespawnReason> Despawned;   // trackers, receiver subscribe (TAC-0-10); no dev subscribers. (rev 17 corrected: "KillPipeline" removed)
        //   (rev 17, code) actual subscribers: WeaponInventory (tracker Forget + child release / orb-zone overlap Exit replay, IMP-PLAYER)
        //   and PlayerDamageReceiver (overlap set / one-shots). KillPipeline does not subscribe. Despawning: MonsterRegistry,
        //   MonsterSpawner (cap count), PickupRegistry (gem/chest counts), PickupSystem (attracted gems); Spawned: MonsterRegistry,
        //   MonsterSpawner, PickupRegistry
        public event Action<SimEntity> Spawned;      // ★ (rev 15) raised last in Spawn; registries (MonsterRegistry, PickupRegistry) add here
        public event Action<SimEntity> Despawning;   // ★ (rev 15) raised first in Despawn; registries remove here (exact reverse of Spawned)
        public IPoolStrategy Pool { get; }            // ★ (rev 15) current run's strategies
        public ITickStrategy Ticks { get; }           // ★ (rev 15) drivers read it every step through Func<ITickStrategy> (5.3)
        public ICollisionStrategy Collisions { get; } // ★ (rev 15)
        public int ActiveCount { get; }               // ★ (rev 15) live entities
        public IReadOnlyList<SimEntity> ActiveEntities { get; } // ★ (rev 15) no particular order — never where R10 order matters
    }

    /// ★ (rev 15) Builds one inactive template per given category and decorated clones of registry prefab overrides (5.6, F11).
    public sealed class PrefabCatalog
    {
        public PrefabCatalog(IReadOnlyDictionary<EntityCategory, Type> entityComponentTypes,
                             IReadOnlyList<IEntityTemplateDecorator> decorators);
        //   composition root passes each category's non-abstract SimEntity subclass (else ArgumentException; null -> ArgumentNullException),
        //   so Simulation never references PLAYER/MONSTER types. A subset of categories is allowed.
        public bool HasTemplate(EntityCategory category);
        public GameObject GetTemplate(EntityCategory category);   // InvalidOperationException when no type was given for category
        public GameObject GetTemplate(EntityCategory category, GameObject prefabOverride);
        //   null -> GetTemplate(category); else an inactive decorated clone (SimEntity type and SpriteRenderer added when missing),
        //   cached per prefab; reusing a prefab for another category, or a prefab carrying another SimEntity type -> InvalidOperationException
    }

    /// ★ (rev 15) BuildIndex order 0: live Monster-category entities in ascending Id (Spawned/Despawning), frozen into the snapshot.
    public sealed class MonsterRegistry : IStepPhaseHandler
    {
        public MonsterRegistry(EntityLifecycle lifecycle, MonsterSnapshotBuffer snapshot);
        public MonsterSnapshotBuffer Snapshot { get; }
        public int Count { get; }
        public IReadOnlyList<SimEntity> Monsters { get; }   // ascending Id
        //   snapshot IsLarge = body.IsLarge || MonsterSizeRules.IsLarge(body.Radius, isBoss: false) (R25)
    }

    public sealed class LifetimeSystem : IStepPhaseHandler { public LifetimeSystem(EntityLifecycle lifecycle); }   // Lifecycle order 10
    public sealed class EntityDespawnCommandHandler : ISimulationCommandHandler { public EntityDespawnCommandHandler(EntityLifecycle lifecycle); }
    [DefaultExecutionOrder(-100)] public sealed class PresentationSync : MonoBehaviour { public void Initialize(EntityLifecycle lifecycle); } // LateUpdate
}
```

- (2회차) 생성 위치·생성 스텝·판정 참여는 5.4의 한 문장을 따른다.
- `Despawned` 구독자는 그 대상 ID의 겹침·쿨다운 기록을 지운다(TAC-0-10). 이벤트는 `OnReleased`와 ID 해제 전에 나가므로 구독자가 `Id`를 읽을 수 있다. (17회차 정정, 웨이브 3 코드 grep) 실제 `Despawned` 구독자는 `WeaponInventory`(무기 트래커 `Forget`, 살아 있는 궤도 원·장판의 겹침 목록에서 제거, 그리고 회수된 것이 궤도 원·장판이면 자식 해제와 그 겹침 목록의 트래커 `OnExit` 재생 — 5.12)와 `PlayerDamageReceiver` 둘이다. **`KillPipeline`은 구독하지 않는다** — 죽은 몬스터는 Resolve order 30에서 `MonsterRegistry.Monsters`를 훑어 스스로 고르고 회수 명령을 넣는다(5.12.1). 레지스트리류(`MonsterRegistry`·`MonsterSpawner`의 상한 집계·`PickupRegistry`·`PickupSystem`의 끌림 목록)는 `Despawning`을 쓴다.
- (15회차, 웨이브 2b 구현 확정 — main b9f8ea5) 위 코드 블록의 `(rev 15, code)` 순서가 구현이다. 2회차 순서 주석의 "registry add"·"registry remove" 자리는 각각 ★ `Spawned`(Spawn 마지막)·★ `Despawning`(Despawn 첫 단계) 이벤트다. 레지스트리(`MonsterRegistry`, 웨이브 3의 `PickupRegistry`)는 `Despawned`가 아니라 `Despawning`으로 제거해 "정확히 역순"을 지킨다. 겹침·쿨다운 기록 소비자는 그대로 `Despawned`다(TAC-0-10 불변).
- (15회차, DECISIONS 2절 2026-11-06) `ReplaceStrategies(pool, ticks, collisions)`는 `DespawnAll` 뒤에만 부른다(기존). 넘겨받은 풀이 현재 풀과 **다른 객체**이면 현재(옛) 풀의 `Clear()`로 비활성 인스턴스를 파괴한 뒤 교체한다. 같은 풀 객체가 유지되면 `Clear()`를 부르지 않는다. 새 풀의 `Prewarm`은 LOOP가 아니라 App이 `PoolPrewarmCounts.PrewarmAll(next.Pool, catalog)`로 한다(5.6, TAC-F8-2 ③).
- (15회차) **[기본값]으로 명문화**: 판 리셋 플래그(private)는 `DespawnAll` 실행 중에만 켜진다. 판 리셋 중 `Spawn` 경로(플레이어 재생성 등)는 문서에 호출 시점이 없어 별도 허용 수단을 두지 않았다 — 판 리셋에서 Lifecycle 밖 `Spawn`이 필요해지면 그 구현자가 보고한다. → (16회차) 이 [기본값]을 유지한다. 플레이어는 `EntityCategory`에 없어 `PrefabCatalog` 템플릿도 `Spawn<T>` 경로도 없으며, TAC-F8-2 ④의 "플레이어 재생성"은 `EntityLifecycle`을 거치지 않는다 — PLAYER가 바디를 직접 만들고 등록한다(5.12 `PlayerEntity`, 5c.3 #1 예외). 그래서 판 리셋 중 Lifecycle 밖 `Spawn` 호출 경로가 없다.
- (15회차) `PrefabCatalog`: 템플릿 이름은 `Template.<category>`, 덮어쓰기 템플릿은 비활성 부모 `PrefabCatalog.Overrides` 아래에서 복제(`Awake`·`OnEnable` 미실행)한 뒤 비활성 상태로 부모에서 떼어 `<prefab>.Template`으로 캐시한다. **[기본값]** 템플릿·`PrefabCatalog.Overrides` 루트를 파괴하는 API는 없다 — 카탈로그는 앱 수명과 같다고 보고 정리 API를 두지 않는다(테스트는 이름으로 찾아 지운다).
- (15회차) `MonsterRegistry` 스냅샷의 `IsLarge`는 `body.IsLarge || MonsterSizeRules.IsLarge(body.Radius, false)`다. `Body.IsLarge`는 몬스터가 보스 여부를 넣어 `MonsterSizeRules.IsLarge(radius, isBoss)`로 채우는 값이고(Core 주석, 5.9), 웨이브 2b 시점에는 `Monster` 클래스가 없어 반지름 규칙을 한 번 더 적용한다(R25).
