### 5.9 충돌 전략 (Core 계약, IMP-COLLISION 구현)

```csharp
namespace Survivor.Core
{
    public enum CollisionStrategyKind { EngineCollider2D = 0, SelfSquaredDistance = 1 }

    public enum CollisionLayer : byte
    { PlayerBody = 0, PlayerMagnet = 1, Monster = 2, PlayerProjectile = 3, PlayerOrb = 4,
      PlayerZone = 5, EnemyProjectile = 6, EnemyZone = 7, Gem = 8, Chest = 9 }

    public enum CollisionChannel : byte
    { PlayerProjectileVsMonster = 0, PlayerOrbVsMonster = 1, PlayerZoneVsMonster = 2,
      PlayerBodyVsMonster = 3, EnemyProjectileVsPlayerBody = 4, EnemyZoneVsPlayerBody = 5,
      PlayerMagnetVsGem = 6, PlayerBodyVsGem = 7, PlayerBodyVsChest = 8 }

    public enum CollisionEventKind : byte { Enter = 0, Exit = 1 }

    /// One Enter/Exit transition of a pair. A is always the body on the channel's A-side layer.
    public readonly struct CollisionEvent : IComparable<CollisionEvent>
    {
        public readonly long Step;
        public readonly CollisionEventKind Kind;
        public readonly CollisionChannel Channel;
        public readonly EntityHandle A;
        public readonly EntityHandle B;
        public CollisionEvent(long step, CollisionEventKind kind, CollisionChannel channel, EntityHandle a, EntityHandle b);
        public int CompareTo(CollisionEvent other);   // (rev 4) Step, Channel, A, B, Kind — the single sort/dispatch/comparison key (TAC-F18-8)
        public override string ToString();            // (rev 4) CSV row: "step,channel,a,b,kind" (same order as CompareTo)
                                                      // (rev 13) channel/kind = enum names, step InvariantCulture, a/b = R15: "120,PlayerBodyVsMonster,1:1,17:3,Enter"
    }

    /// (rev 13) Single owner of the nine channels and their A/B layers (R5). The A layer is the first layer of the channel name.
    public static class CollisionChannelTable
    {
        public const int ChannelCount = 9;    // CollisionChannel values 0..8
        public static bool TryGetChannel(CollisionLayer x, CollisionLayer y, out CollisionChannel ch, out bool xIsA); // either order; false when no channel
        public static void GetLayers(CollisionChannel channel, out CollisionLayer a, out CollisionLayer b);           // throws ArgumentOutOfRangeException for an undefined channel
    }

    /// (rev 13) Collision layer -> Unity layer (R30).
    public static class CollisionLayerMapping
    {
        public const int FirstUnityLayer = 8;
        public static int UnityLayerOf(CollisionLayer layer);   // 8..17; throws ArgumentOutOfRangeException for an undefined layer
        public static int MaskOf(CollisionLayer layer);         // single bit: 1 << UnityLayerOf(layer)
    }

    /// Mutable circle owned by an entity; strategies read Position/Radius during CollectEvents.
    public sealed class CollisionBody
    {
        public EntityHandle Id;
        public CollisionLayer Layer;
        public Vector2 Position;
        public float Radius;
        public bool IsLarge;          // (rev 3) MonsterSizeRules.IsLarge(Radius, isBoss) for monsters, false otherwise
        public GameObject Host;       // PhysicsBodySynchronizer toggles CircleCollider2D/Rigidbody2D on it
        public int StrategySlot = -1; // private to the active collision strategy
        public int PhysicsSlot = -1;  // (rev 2) private to PhysicsBodySynchronizer
        public bool DrivesTransform;  // set by PhysicsBodySynchronizer when engine simulation owns the transform
                                      // (rev 18) = Enabled && Link.Sink != null (collider on and an EngineCollisionStrategy registered as sink)
    }

    /// Polling collision detection; both strategies must emit the same sorted event set.
    public interface ICollisionStrategy
    {
        CollisionStrategyKind Kind { get; }
        void Register(CollisionBody body);      // participates from the next CollectEvents
        void Unregister(CollisionBody body);    // drops its overlaps silently (no Exit)
        void NotifyRadiusChanged(CollisionBody body);
        //   (rev 17) Engine: forwards to PhysicsBodySynchronizer.NotifyRadiusChanged; CircleCollider2D.radius is updated at the next
        //   SyncAllForSimulation (C7). Self: validates the registration only (Radius is read every CollectEvents). Callers: PlayerEntity
        //   (magnet, P3) and OrbitWeapon (orbs after an upgrade) — 5.12, 5.14
        IReadOnlyList<CollisionEvent> CollectEvents(long step); // sorted ascending; list reused until next call
        void Clear();                           // run reset
    }

    /// (rev 2) Registration hook for Unity physics bodies; implemented by Strategies, called by EntityLifecycle.
    public interface IPhysicsBodyRegistry
    {
        void Register(CollisionBody body);      // teleports rb/transform to body.Position; enables collider if required (R19)
        void Unregister(CollisionBody body);    // disables collider (no Exit callbacks: callbacksOnDisable = false)
    }
}
```

- (17회차, PM 판정 — 구현 보정) **엔진판 ③ 순변화 필터**(TAC-F18-4 개정). 콜백을 그대로 이벤트로 내지 않고, 쌍(`(Channel, A, B)`)마다 이번 `Simulate` 전의 활성 여부를 기억한 뒤 콜백을 적용해, 끝의 활성 여부가 달라진 쌍만 Enter/Exit 1개로 낸다. 같은 `Simulate` 안에 Enter·Exit가 연달아 온 쌍(이번 스텝 등록된 바디가 상대의 직전 스텝 위치와 겹친 경우 — 실측 seed 1 스텝 340, seed 3 스텝 39)은 이벤트 없이 `CancelledTransitionCount`만 늘린다. 쌍끼리는 독립이라 쌍 간 순서는 결과에 영향이 없고, 마지막에 `CollisionEvent.CompareTo`로 정렬한다. 원인 추정(Box2D가 이동 전에 접촉을 갱신)은 **미확정**이다. 필터 뒤 의미는 자체판(직전 스텝 집합과의 차분)과 같다 — E1 골격 3배치 일치(5c.5 E1). (23회차 수정) 이 필터는 아래 "R5 게이트"에 **포함**된다 — 순변화는 이제 Box2D 접촉 상태가 아니라 **R5 상태의** 스텝 시작 → 끝 변화다. 위 실측 두 사례는 끝 위치에서 R5가 거짓이라(거리 > 반지름 합) R5 게이트만으로 이벤트가 없다. `_touched`(스텝 시작의 Box2D 상태)는 `CancelledTransitionCount` 진단에만 남는다.
- (23회차 수정, DECISIONS 4절 "엔진 충돌판은 Box2D 접촉 쌍을 후보로만 쓰고 보고는 R5 정확 판정으로 낸다 (2026-11-12)") **엔진판 이벤트 생성 = 후보 쌍의 R5 상태 순변화(R5 게이트).** 보고되는 겹침 상태의 단일 소유자는 두 전략 모두 R5 `CircleMath.Overlaps`이고, Box2D는 후보 탐색(넓은 판정)만 맡는다. 이벤트 정렬·채널·`CollisionEvent` 형식·`ICollisionStrategy`·생성자는 바뀌지 않는다. 발견 경위: 웨이브 7 IMP-PLAYER 36조합에서 엔진 18조합만 다른 해시, 첫 차이 스텝 1125에 엔진판만 `Enter PlayerProjectileVsMonster 146:4 305:2`(d² 0.490000129 > r² 0.48999998 — Box2D 허용오차), 자체판은 1126(PM 재현 2026-11-12).   - **상태(★ private, 이름은 이 회차가 정함)**: `_activePairs`를 `HashSet<PairSet.Key>`에서 ★ `Dictionary<PairSet.Key, BodyPair>`로(Box2D가 접촉 중이라고 아는 쌍 + 두 바디 참조). ★ `_reportedPairs : Dictionary<PairSet.Key, BodyPair>` — Enter를 보고했고 아직 Exit을 보고하지 않은 쌍(자체판 `PairSet`의 직전 스텝 집합에 해당). ★ `_candidates : Dictionary<PairSet.Key, BodyPair>`(스텝 스크래치). ★ `private readonly struct BodyPair { internal readonly CollisionBody A; internal readonly CollisionBody B; }`(A = 채널 A측). `QueuedCallback`에 A·B 바디 필드를 더한다. 보고 집합에 `PairSet`(internal)을 쓰지 않는다 — 자체판용 "직전/수집 목록 차분" 클래스라 `Contains`가 없다(`PairSet.cs:20~102` 멤버는 `Begin`·`Add`·`Commit`·`RemoveBody`·`Clear`).
  - **바디 조회 경로(코드 확인)**: `PairSet.Key`는 `(Channel, A, B)` ID만 담는다(`PairSet.cs:124~135`). 바디는 `OnTrigger`에서만 얻는다 — `self`는 `CollisionBodyLink.Body`(`CollisionBodyLink.cs:20·28`, 동기화기가 `Register` 때 연결 `PhysicsBodySynchronizer.cs:84`), 상대는 `_bodies.TryGetBody(other, out otherBody)`(동기화기 `_byCollider`, `PhysicsBodySynchronizer.cs:199~209`; `EngineCollisionStrategy.cs:212`). 큐에는 A측 호스트의 콜백만 들어가므로(C6, `EngineCollisionStrategy.cs:223~229`) `BodyPair(self, otherBody)`다. 콜백 없이 이어지는 쌍은 저장된 참조를 쓴다 — ID → 바디 조회 표는 새로 두지 않는다. 저장된 참조는 `Unregister`가 그 ID의 쌍을 두 집합에서 지우므로 항상 이 전략에 등록된 바디다. 풀 재사용으로 같은 `CollisionBody` 객체가 새 ID를 받는 것은 해제 뒤다(`EngineCollisionStrategyTests.Unregister_WhileOverlapping_NoExit_ReusedHostWithNewId_Enters`).
  - **`CollectEvents(step)`**: ① `SyncAllForSimulation()` ② `Physics2D.Simulate(StepSeconds)`(둘 다 불변) ③ 큐 접기 — 콜백 순서대로 Enter면 `_activePairs[key] = bodies`, Exit면 `Remove(key)`; `_touched`로 `CancelledTransitionCount`를 기존대로 센다 ④ 후보 = `_activePairs` ∪ 이번 스텝 큐의 쌍 ∪ `_reportedPairs`를 `_candidates`에 모은다(키 중복 제거). `_reportedPairs`를 넣는 이유: Box2D가 Exit을 냈지만 R5가 아직 참인 쌍(허용오차가 반대로 작용하는 이탈)도 R5가 거짓이 되는 스텝까지 매 스텝 다시 보고 **그 스텝에** Exit을 낸다 — 자체판과 같은 스텝. 큐의 쌍을 넣는 이유: 한 `Simulate` 안 Enter·Exit 상쇄 쌍도 끝 위치의 R5로 판정한다 ⑤ 후보마다 `bool exact = CircleMath.Overlaps(p.A.Position, p.A.Radius, p.B.Position, p.B.Radius)`(R5 단일 소유 함수, 자체판과 **같은 함수·같은 인자 순서** — A측 먼저, `SelfCollisionStrategy.cs:261·279·331`), `bool reported = _reportedPairs.ContainsKey(key)`. `exact && !reported` → Enter 추가 + `_reportedPairs.Add`, `!exact && reported` → Exit 추가 + `Remove`, 같으면 이벤트 없음. `_activePairs`에 있는데 `exact`가 거짓이면 ★ `BoundaryDeferredCount++` ⑥ `_events.Sort(EventComparison)`(불변). 쌍끼리 독립이라 `_candidates` 순회 순서는 결과에 영향이 없다(이벤트는 정렬되고 집합 갱신은 쌍별).

namespace Survivor.Simulation
{
    public interface ICollisionEventHandler { void Handle(in CollisionEvent e, in StepContext context); }
    public sealed class CollisionEventRouter : IStepPhaseHandler   // Collision order 0 collects, Resolve order 0 dispatches in sorted order

- (17회차, B8·U28 부분 확인) `Physics2D.autoSyncTransforms`는 6000.3에서 deprecated(CS0618)이지만 동작하므로, 구현은 `#pragma warning disable CS0618`로 감싸 기존 API를 그대로 쓴다. 나머지 기존 `Physics2D` API는 웨이브 3 테스트에서 동작했다(6.2 U28, 미확정 유지).
- (17회차, PM 판정 [기본값]) **반지름 변경 경로**: 엔진판 `NotifyRadiusChanged`는 `PhysicsBodySynchronizer.NotifyRadiusChanged`로 넘기고, 다음 `SyncAllForSimulation`(또는 `SyncMonstersForQuery`·콜라이더 켜기)에서 `CircleCollider2D.radius`를 갱신한다(C7). 동기화기는 `body.Radius`와 마지막 적용값이 달라도 갱신하므로 알림 누락에 안전한 쪽이다. 부르는 쪽은 PLAYER 두 곳이다(5.12 자석 각주, 5.14).
- (18회차, 웨이브 4c 결함 B 수정 — PM 수용, 설계 변경 아님) **`Body.DrivesTransform`의 의미 확정**: `DrivesTransform = Enabled && Link.Sink != null` — 콜라이더가 켜져 있고 엔진판(`EngineCollisionStrategy`)이 그 바디의 링크 싱크로 등록돼 있을 때만 참이다. 동기화기가 `Register`·`Unregister`·`ApplyRequirements` 때마다 다시 계산하고, 싱크를 붙이고 떼는 엔진판의 `Register`·`Unregister`·`Clear`가 ★ internal `PhysicsBodySynchronizer.RefreshDrivesTransform(CollisionBody)`를 부른다. 자체 충돌 모드에서는 싱크가 없으므로 항상 거짓이다. 배경: 종전 `Register`가 이 값을 항상 거짓으로 두어 `PresentationSync`가 엔진 모드 바디의 `transform`을 매 프레임 썼다. Box2D 적분값과 ~1 ULP 다른 그 쓰기를 다음 `Simulate`가 **텔레포트로 취급해 그 스텝의 `MovePosition`을 버렸고**, 엔진판이 바디를 한 스텝 늦게 봐 자체판과 어긋났다(불일치가 전부 프레임당 스텝 수 6의 배수에서 났다). 이것이 U10의 "`transform` 직접 쓰기는 텔레포트" 확인이다(6.2). 회귀 테스트 `Tests/PlayMode/Collision/EnginePresentationWriteTests`(2). 위치를 쓰는 표현 코드 둘 — `PresentationSync`(LOOP)와 `PlayerPresenter`(App, 5.17) — 모두 `Body.DrivesTransform`이면 위치 쓰기를 생략한다(5c.3 #6).
