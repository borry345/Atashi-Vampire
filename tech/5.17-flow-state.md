### 5.17 흐름 상태 (App, IMP-FLOW) (2회차 신규)

```csharp
namespace Survivor.App
{
    public enum FlowState { Title, Playing, Paused, LevelUp, Result, DataError }

    public sealed class GameFlowController : MonoBehaviour
    {
        // Update: polls clock.CloseSequence -> LevelUp (open round) / Death, Clear (result + best record) / Error (error view) / StopAtStep (no-op; dev handles)
        // ESC: Playing -> Paused(return Playing); LevelUp -> Paused(return LevelUp); Paused -> return state
        // OnApplicationFocus(false): Playing && Rules.AutoPauseOnFocusLoss -> Paused (TAC-F8-9)
        // UI views raise events (UI never references App); this class subscribes.
        // (rev 8) Best record is saved only on the Death/Clear result path (TAC-F8-3). Error (AbortStep) and forced quit (Alt+F4 etc.) never save.
        // (rev 18, code) Paused, RunReset and SingleStepDone closes are ignored too; one new close is handled per poll.
        public FlowState State { get; }              // ★ (rev 18) Title at start (DataError when the boot data is not playable)
        public FlowState ResumeState { get; }        // ★ (rev 18) Playing or LevelUp; meaningful while State == Paused
        public bool ShowsRuntimeError { get; }       // ★ (rev 18) State == DataError because a step aborted (data itself playable)
        public RunOutcome LastOutcome { get; }       // ★ (rev 18) outcome of the latest finished run (RunOutcome.None before the first)
        public void StartRun();      // ★ (rev 18) title "시작" / result "재시작" / IDevModeHost.StartNewRun: GameBootstrap.ResetRun -> HUD -> Playing -> clock.Open().
                                     //   Refused (WARN log) when the data is not playable; no state check otherwise
        public void ExitToTitle();   // ★ (rev 18) pause "타이틀로 나가기" / result "타이틀" / error "타이틀로": ResetRun without Open, nothing recorded
        public void TogglePause();   // ★ (rev 18) ESC: handles a pending close first, then Playing -> Close(Paused) -> Paused; LevelUp -> Paused (round kept);
                                     //   Paused -> return state (Playing reopens the gate); other states ignore it
        public void Quit();          // ★ (rev 18) title "종료": Application.Quit(), no record
        // internal Initialize(GameBootstrap host, TitleView, HudView, LevelUpView, PauseMenuView, ResultView, DataErrorView, UiInputHandler, BestRecordStore)
        //   — composition root only; subscribes to every view event and shows Title or DataError (TAC-F8-4)
    }

    /// ★ (rev 18) Composition root and IDevModeHost (5.15). [DefaultExecutionOrder(-5000)]. Main.unity holds this component, the camera and Global Light 2D.
    public sealed class GameBootstrap : MonoBehaviour, IDevModeHost
    {
        public GameFlowController Flow { get; }      // ★ (rev 18) flow controller (screen state)
        // + the 15 IDevModeHost members exactly as in 5.15 (implementation notes there)
        // Awake: SimulationSettingsApplier.Apply -> PhysicsSettingsApplier.Apply -> DisplaySetup.Apply -> data load (5.11.0) -> asset key resolver
        //   (5.11) -> Global Light 2D if the scene has none (5b) -> Assemble (5c.3a wiring, 5c.3 #9 orders, drivers) -> BuildPresentation
        //   (camera size 9 + CameraFollow, Background + BackgroundTiler, PlayerPresenter on the player host) -> BuildUi (canvas, EventSystem with
        //   InputSystemUIInputModule when none, six views, UiInputHandler, BestRecordStore, GameFlowController)
    }

    /// ★ (rev 18) Builds a run's Strategies instances (only App references Strategies, 5c.1). Undefined kinds -> ArgumentOutOfRangeException.
    public sealed class StrategyFactory
    {
        public StrategyFactory(IStepClockReader clock, MonsterSpatialIndex index, MonsterSnapshotBuffer snapshot, PhysicsBodySynchronizer bodies); // null -> ArgumentNullException
        public static int MonsterLayerMask { get; }  // CollisionLayerMapping.MaskOf(CollisionLayer.Monster) (R30), for Physics2DTargetProvider
        public IPoolStrategy CreatePool(PoolStrategyKind kind);
        public ITickStrategy CreateTick(TickStrategyKind kind);
        public ICollisionStrategy CreateCollision(CollisionStrategyKind kind, CollisionDetectionMode2D engineMode);
        public ITargetProvider CreateTarget(TargetProviderKind kind);
        public RunStrategies Create(in StrategySelection selection, IPoolStrategy currentPool,
                                    Func<ICollisionStrategy, ICollisionStrategy> collisionDecorator);
        //   currentPool is reused when currentPool.Kind == selection.Pool (no Clear, no prewarm, 5.10); otherwise a new pool.
        //   collisionDecorator (null = none) wraps the collision strategy (dev logging, IDevModeHost.NextRunCollisionDecorator);
        //   returning null -> InvalidOperationException. Template decorators are NOT made here (see notes below)
    }

    [DefaultExecutionOrder(0)]   public sealed class CameraFollow : MonoBehaviour     // ★ (rev 18) LateUpdate: camera = target() exactly (TAC-F1-6)
    { public void Initialize(Func<Vector2> target); }
    [DefaultExecutionOrder(100)] public sealed class BackgroundTiler : MonoBehaviour  // ★ (rev 18) Tiled checker that snaps to the camera (TAC-F8-11, U29)
    {
        public const float TileSize = 2f;
        public Vector2 CoveredSize { get; }          // view + one tile of margin per side
        public void Initialize(Transform cameraTransform, Vector2 viewSize);
    }
    [DefaultExecutionOrder(-100)] public sealed class PlayerPresenter : MonoBehaviour // ★ (rev 18) player circle (sorting 10) on a child "Player.Visual"
    { public void Initialize(PlayerEntity player); }  // LateUpdate: host position = player.Position unless player.Body.DrivesTransform (5c.3 #6, 5.9)
    // EntityPresenter (pooled entity visuals, [DefaultExecutionOrder(-50)]): 5.11 "(18회차) 소비자"
}
```

- (18회차, 웨이브 4b 구현 확정) 위 블록은 `Scripts/App/`·`Scripts/UI/`의 실제 공개 면이다. UI 뷰의 버튼 대행 공개 메서드(`RequestStart`·`RequestQuit`·`RequestResume`·`RequestExitToTitle`·`RequestRestart`·`RequestTitle`, 이벤트를 내는 것과 같음)와 읽기 속성(`HudView.TimeText` 등, `ResultView.HeaderText` 등, `DataErrorView.IssueCount`·`BodyText`·`TitleButtonVisible`, `LevelUpView.LabelOf`, `TitleView.StartEnabled`)은 테스트가 화면 내용을 읽는 용도라 블록에서 생략했다(코드 주석 참조). `FlowState` 열거형은 별도 파일 `FlowState.cs`다.
- (18회차) 테스트 훅: `GameBootstrap.DataRootOverride`·`BestRecordKeyOverride`(internal static, 씬 로드 전 설정)와 internal 접근자(`Pipeline`·`CommandBuffer`·`Router`·`Player`·`Presenter` 등), `GameFlowController`의 internal 뷰 접근자·`HandleFocusChanged` — `AppAssemblyInfo.cs`의 `InternalsVisibleTo("Survivor.Tests.EditMode")`·`("Survivor.Tests.PlayMode")`.
- (18회차) **`StrategyFactory.Create`의 데코레이터 인자와 템플릿 데코레이터는 다른 것이다.** `Create`의 `collisionDecorator`는 판의 충돌 전략 하나를 감싸는 함수(`IDevModeHost.NextRunCollisionDecorator`, 개발자 충돌 로그)다. 템플릿 데코레이터 3종(`PoolTemplateDecorator`·`SelfTickTemplateDecorator`·`CollisionTemplateDecorator`)은 `GameBootstrap`이 앱 수명의 `PrefabCatalog`를 만들 때 **선택한 전략과 무관하게 셋 다** 넘긴다(TAC-F9-7) — 카탈로그는 앱 수명이고 전략은 판마다 바뀌기 때문이다. `StrategyFactory`는 템플릿 데코레이터를 만들지 않는다.
- (18회차, PM 수용 — 설계 변경 아님) ① **풀 재사용**: 다음 판의 풀 종류가 지금과 같으면 `Create`가 지금 풀 인스턴스를 그대로 돌려주고(`ReplaceStrategies`에서 같은 객체라 `Clear()` 없음), 다를 때만 새 풀 + `PrewarmAll`이다(5.10, TAC-F8-2 ③). ② **판 중 런타임 오류**(`AbortStep`, 게이트 `Error`)는 `FlowState.DataError` 상태로 표시한다 — `DataErrorView.ShowRuntimeError`(타이틀로 버튼), `ShowsRuntimeError == true`, 기록 없음. 데이터 오류(부팅)와는 `ShowsRuntimeError`로 구분한다. ③ **판 시드** = `FixedSeed ?? (ulong)DateTime.UtcNow.Ticks`, 판 경계에서 1회(`RunSession`의 `nextSeed`, TAC-0-2의 일반 플레이 예외). ④ **HUD** = 시간(스텝)·레벨·체력/최대 체력·경험치/필요 경험치, `GameFlowController.Update`에서 HUD가 보일 때 매 프레임 갱신. ⑤ **`Main.unity`** = `Main Camera`(URP 2D 렌더러)·`Global Light 2D`·`GameBootstrap` 오브젝트 1개뿐이고, 나머지(`Player`·`Player.Magnet`·`Simulation`·`Input`·`Background`·`UI`·`EventSystem`)는 전부 런타임 생성이다. `GameBootstrap`은 `Camera.main`이 없으면 카메라를, Global Light 2D가 없으면 조명을 코드로 만든다. ⑥ **로그 영역**: App은 internal `AppLog`(`[HH:mm:ss.fff] [LEVEL] [Flow] message`)를 쓴다(판 시작·타이틀 복귀·레벨업 회차·결과·런타임 오류·데이터 불가·창 크기·공급자 교체 요청, 프레임마다 없음). `[Player]`·`[Flow]`·`[Loop]` 영역 정리는 5c.4 끝 정리 후보다.
